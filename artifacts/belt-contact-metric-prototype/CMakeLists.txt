cmake_minimum_required(VERSION 3.22)
project(contact_metric_prototype C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(mode RANGE 0 1)
  add_executable(metric_${mode} experiment.c)
  target_compile_features(metric_${mode} PRIVATE c_std_17)
  target_compile_definitions(metric_${mode} PRIVATE EXPERIMENT_METRIC=${mode})
endforeach()
add_executable(room_metric src/dango_belt_room_tests.c)
target_compile_definitions(room_metric PRIVATE DANGO_BELT_ROOM_TRACE=1)
add_executable(wall_metric src/dango_belt_room_wall_tests.c)
foreach(target room_metric wall_metric)
  target_compile_features(${target} PRIVATE c_std_17)
  target_compile_definitions(${target} PRIVATE EXPERIMENT_METRIC=1)
endforeach()
