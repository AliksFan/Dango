cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(dango_endpoint_selection LANGUAGES C)
if(NOT DEFINED DANGO_ENDPOINT_GENERATED)
    message(FATAL_ERROR "A fresh verified endpoint source copy is required")
endif()
get_filename_component(DANGO_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX2D_DOCS OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_VALIDATE ON CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${DANGO_ENDPOINT_GENERATED}/box2d" box2d EXCLUDE_FROM_ALL)
target_compile_definitions(box2d PRIVATE DANGO_RECEIPT_HISTORY_CANDIDATE)
target_include_directories(box2d PRIVATE "${DANGO_NATIVE}/receipt-history")
file(STRINGS "${DANGO_NATIVE}/CMakeLists.txt" repair_line REGEX "^set\\(dango_material_repair_final_definitions ")
string(REGEX REPLACE "^set\\(dango_material_repair_final_definitions (.*)\\)$" "\\1" repair_definitions "${repair_line}")
separate_arguments(repair_definitions)
add_library(dango_material_endpoint_selection_candidate SHARED candidate.c "${DANGO_NATIVE}/receipt-history/capture.c" "${DANGO_NATIVE}/dango_material_forces.c")
add_executable(dango_endpoint_receipt_tests "${DANGO_NATIVE}/receipt-history/tests.c" "${DANGO_NATIVE}/receipt-history/capture.c" "${DANGO_NATIVE}/dango_material_forces.c")
add_executable(dango_endpoint_fixed_tests tests.c "${DANGO_NATIVE}/receipt-history/capture.c" "${DANGO_NATIVE}/dango_material_forces.c")
foreach(target IN ITEMS dango_material_endpoint_selection_candidate dango_endpoint_receipt_tests dango_endpoint_fixed_tests)
    target_compile_features(${target} PRIVATE c_std_17)
    target_compile_definitions(${target} PRIVATE ${repair_definitions} DANGO_RECEIPT_HISTORY_CANDIDATE DANGO_MATERIAL_EXPERIMENT_MODE=15 DANGO_MATERIAL_CONTACT_WIDE_GROUND_CANDIDATE DANGO_MATERIAL_CONTACT_ZERO_TO_SLIDE_CANDIDATE)
    target_include_directories(${target} BEFORE PRIVATE "${DANGO_ENDPOINT_GENERATED}/adapter" "${DANGO_NATIVE}/receipt-history" "${DANGO_NATIVE}" "${DANGO_ENDPOINT_GENERATED}/box2d/src")
    target_link_libraries(${target} PRIVATE box2d::box2d)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    endif()
endforeach()
set_target_properties(dango_material_endpoint_selection_candidate PROPERTIES PREFIX "")
if(MSVC)
    target_link_options(dango_material_endpoint_selection_candidate PRIVATE /Brepro)
endif()
