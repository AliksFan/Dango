cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)

project(dango_box2d_bridge LANGUAGES C)

if(NOT DEFINED BOX2D_SOURCE_DIR)
    message(FATAL_ERROR "BOX2D_SOURCE_DIR must point to the pinned official Box2D source tree")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX2D_DOCS OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_VALIDATE ON CACHE BOOL "" FORCE)
set(BOX2D_COMPILE_WARNING_AS_ERROR OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_subdirectory("${BOX2D_SOURCE_DIR}" box2d EXCLUDE_FROM_ALL)

add_library(dango_box2d SHARED dango_box2d_bridge.c)
add_library(dango_collision_review SHARED dango_collision_review.c dango_box2d_bridge.c)
target_link_libraries(dango_collision_review PRIVATE box2d::box2d)
target_compile_features(dango_collision_review PRIVATE c_std_17)
set_target_properties(dango_collision_review PROPERTIES PREFIX "")
add_library(dango_physics_module SHARED dango_box2d_bridge.c dango_collision_review.c)
target_link_libraries(dango_physics_module PRIVATE box2d::box2d)
target_compile_features(dango_physics_module PRIVATE c_std_17)
set_target_properties(dango_physics_module PROPERTIES PREFIX "" UNITY_BUILD ON UNITY_BUILD_BATCH_SIZE 2)
add_executable(dango_self_constraints_tests dango_self_constraints_tests.c)
add_executable(dango_spring_balance_tests dango_spring_balance_tests.c)
add_executable(dango_material_idle_diagnostics dango_material_idle_diagnostics.c dango_material_forces.c)
foreach(capture_mode IN ITEMS plain observed)
    add_executable(dango_material_contact_${capture_mode} dango_material_contact_diagnostics.c dango_material_forces.c)
    target_link_libraries(dango_material_contact_${capture_mode} PRIVATE box2d::box2d)
    target_compile_features(dango_material_contact_${capture_mode} PRIVATE c_std_17)
    if(capture_mode STREQUAL "observed")
        target_compile_definitions(dango_material_contact_${capture_mode} PRIVATE DANGO_CONTACT_OBSERVER_BUILD)
    endif()
    if(MSVC)
        target_compile_options(dango_material_contact_${capture_mode} PRIVATE /W4 /WX)
    endif()
endforeach()
foreach(observer_test IN ITEMS dango_constraint_observer_tests dango_constraint_observer_tests_off)
    add_executable(${observer_test} dango_constraint_observer_tests.c)
    target_link_libraries(${observer_test} PRIVATE box2d::box2d)
    target_compile_features(${observer_test} PRIVATE c_std_17)
    if(observer_test STREQUAL "dango_constraint_observer_tests_off")
        target_compile_definitions(${observer_test} PRIVATE DANGO_CONSTRAINT_OBSERVER_TEST_DISABLED)
    endif()
    if(MSVC)
        target_compile_options(${observer_test} PRIVATE /W4 /WX)
    endif()
endforeach()
add_executable(dango_material_contact_solver_tests dango_material_contact_solver_tests.c)
target_link_libraries(dango_material_contact_solver_tests PRIVATE box2d::box2d)
target_compile_features(dango_material_contact_solver_tests PRIVATE c_std_17)
if(MSVC)
    target_compile_options(dango_material_contact_solver_tests PRIVATE /W4 /WX)
endif()
add_executable(dango_material_contact_block_tests dango_material_contact_block_tests.c)
target_link_libraries(dango_material_contact_block_tests PRIVATE box2d::box2d)
target_compile_features(dango_material_contact_block_tests PRIVATE c_std_17)
if(MSVC)
    target_compile_options(dango_material_contact_block_tests PRIVATE /W4 /WX)
endif()
add_library(dango_material_contact_candidate SHARED dango_box2d_bridge.c dango_collision_review.c dango_material_forces.c)
target_link_libraries(dango_material_contact_candidate PRIVATE box2d::box2d)
target_compile_features(dango_material_contact_candidate PRIVATE c_std_17)
target_compile_definitions(dango_material_contact_candidate PRIVATE DANGO_MATERIAL_CONTACT_CANDIDATE DANGO_MATERIAL_EXPERIMENT_MODE=7)
set_target_properties(dango_material_contact_candidate PROPERTIES PREFIX "")
add_executable(dango_material_candidate_diagnostics dango_material_candidate_diagnostics.c dango_material_forces.c)
target_link_libraries(dango_material_candidate_diagnostics PRIVATE box2d::box2d)
target_compile_features(dango_material_candidate_diagnostics PRIVATE c_std_17)
target_compile_definitions(dango_material_candidate_diagnostics PRIVATE DANGO_MATERIAL_CONTACT_CANDIDATE)
add_executable(dango_material_contact_latch_tests dango_material_contact_latch_tests.c dango_material_forces.c)
target_link_libraries(dango_material_contact_latch_tests PRIVATE box2d::box2d)
target_compile_features(dango_material_contact_latch_tests PRIVATE c_std_17)
if(MSVC)
    target_compile_options(dango_material_contact_candidate PRIVATE /W4 /WX)
    target_link_options(dango_material_contact_candidate PRIVATE /Brepro)
    target_compile_options(dango_material_candidate_diagnostics PRIVATE /W4 /WX)
    target_compile_options(dango_material_contact_latch_tests PRIVATE /W4 /WX)
endif()
add_library(dango_material_contact_block_candidate SHARED dango_box2d_bridge.c dango_collision_review.c dango_material_forces.c)
target_link_libraries(dango_material_contact_block_candidate PRIVATE box2d::box2d)
target_compile_features(dango_material_contact_block_candidate PRIVATE c_std_17)
target_compile_definitions(dango_material_contact_block_candidate PRIVATE DANGO_MATERIAL_CONTACT_CANDIDATE DANGO_MATERIAL_CONTACT_BLOCK_CANDIDATE DANGO_MATERIAL_EXPERIMENT_MODE=7)
set_target_properties(dango_material_contact_block_candidate PROPERTIES PREFIX "")
add_executable(dango_material_contact_block_diagnostics dango_material_candidate_diagnostics.c dango_material_forces.c)
target_link_libraries(dango_material_contact_block_diagnostics PRIVATE box2d::box2d)
target_compile_features(dango_material_contact_block_diagnostics PRIVATE c_std_17)
target_compile_definitions(dango_material_contact_block_diagnostics PRIVATE DANGO_MATERIAL_CONTACT_CANDIDATE DANGO_MATERIAL_CONTACT_BLOCK_CANDIDATE)
add_executable(dango_material_contact_block_latch_tests dango_material_contact_block_latch_tests.c dango_material_forces.c)
target_link_libraries(dango_material_contact_block_latch_tests PRIVATE box2d::box2d)
target_compile_features(dango_material_contact_block_latch_tests PRIVATE c_std_17)
if(MSVC)
    target_compile_options(dango_material_contact_block_candidate PRIVATE /W4 /WX)
    target_link_options(dango_material_contact_block_candidate PRIVATE /Brepro)
    target_compile_options(dango_material_contact_block_diagnostics PRIVATE /W4 /WX)
    target_compile_options(dango_material_contact_block_latch_tests PRIVATE /W4 /WX)
endif()
foreach(material_mode IN ITEMS area bending both both_phase both_phase_no_spin both_phase_configured_friction both_phase_projected_radial)
    if(material_mode STREQUAL "area")
        set(material_bits 1)
    elseif(material_mode STREQUAL "bending")
        set(material_bits 2)
    elseif(material_mode STREQUAL "both_phase")
        set(material_bits 7)
    elseif(material_mode STREQUAL "both_phase_no_spin")
        set(material_bits 15)
    elseif(material_mode STREQUAL "both_phase_configured_friction")
        set(material_bits 23)
    elseif(material_mode STREQUAL "both_phase_projected_radial")
        set(material_bits 39)
    else()
        set(material_bits 3)
    endif()
    add_library(dango_material_${material_mode} SHARED dango_box2d_bridge.c dango_collision_review.c dango_material_forces.c)
    target_link_libraries(dango_material_${material_mode} PRIVATE box2d::box2d)
    target_compile_features(dango_material_${material_mode} PRIVATE c_std_17)
    target_compile_definitions(dango_material_${material_mode} PRIVATE DANGO_MATERIAL_EXPERIMENT_MODE=${material_bits})
    set_target_properties(dango_material_${material_mode} PROPERTIES PREFIX "")
    if(MSVC)
        target_compile_options(dango_material_${material_mode} PRIVATE /W4 /WX)
        target_link_options(dango_material_${material_mode} PRIVATE /Brepro)
    endif()
endforeach()
add_executable(dango_material_force_tests dango_material_force_tests.c dango_material_forces.c)
add_executable(dango_material_integration_tests dango_material_integration_tests.c dango_material_forces.c)
target_link_libraries(dango_material_integration_tests PRIVATE box2d::box2d)
target_compile_features(dango_material_integration_tests PRIVATE c_std_17)
target_link_libraries(dango_material_force_tests PRIVATE box2d::box2d)
target_compile_features(dango_material_force_tests PRIVATE c_std_17)
target_link_libraries(dango_material_idle_diagnostics PRIVATE box2d::box2d)
target_compile_features(dango_material_idle_diagnostics PRIVATE c_std_17)
target_link_libraries(dango_spring_balance_tests PRIVATE box2d::box2d)
target_compile_features(dango_spring_balance_tests PRIVATE c_std_17)
target_link_libraries(dango_self_constraints_tests PRIVATE box2d::box2d)
target_compile_features(dango_self_constraints_tests PRIVATE c_std_17)
target_link_libraries(dango_box2d PRIVATE box2d::box2d)
target_compile_features(dango_box2d PRIVATE c_std_17)
target_compile_definitions(dango_box2d PRIVATE DANGO_BOX2D_BUILD)
set_target_properties(dango_box2d PROPERTIES
    OUTPUT_NAME "dango_box2d"
    PREFIX ""
    WINDOWS_EXPORT_ALL_SYMBOLS OFF
)

if(MSVC)
    target_compile_options(dango_collision_review PRIVATE /W4 /WX)
    target_compile_options(dango_physics_module PRIVATE /W4 /WX)
    target_compile_options(dango_spring_balance_tests PRIVATE /W4 /WX)
    target_compile_options(dango_material_idle_diagnostics PRIVATE /W4 /WX)
    target_compile_options(dango_material_force_tests PRIVATE /W4 /WX)
    target_compile_options(dango_material_integration_tests PRIVATE /W4 /WX)
    target_compile_options(dango_self_constraints_tests PRIVATE /W4 /WX)
    target_compile_options(dango_box2d PRIVATE /W4 /WX)
    target_link_options(dango_box2d PRIVATE /Brepro)
    target_link_options(dango_physics_module PRIVATE /Brepro)
endif()
