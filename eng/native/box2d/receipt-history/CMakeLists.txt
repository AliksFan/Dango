cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(dango_receipt_history LANGUAGES C)
if(NOT DEFINED DANGO_RECEIPT_GENERATED)
    message(FATAL_ERROR "DANGO_RECEIPT_GENERATED must point to a new verified generated copy")
endif()
get_filename_component(DANGO_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX2D_DOCS OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_VALIDATE ON CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${DANGO_RECEIPT_GENERATED}/box2d" box2d EXCLUDE_FROM_ALL)
target_compile_definitions(box2d PRIVATE DANGO_RECEIPT_HISTORY_CANDIDATE)
target_include_directories(box2d PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
file(STRINGS "${DANGO_NATIVE}/CMakeLists.txt" repair_line REGEX "^set\\(dango_material_repair_final_definitions ")
string(REGEX REPLACE "^set\\(dango_material_repair_final_definitions (.*)\\)$" "\\1" repair_definitions "${repair_line}")
separate_arguments(repair_definitions)
add_library(dango_material_receipt_history_candidate SHARED candidate.c capture.c "${DANGO_NATIVE}/dango_material_forces.c")
target_compile_features(dango_material_receipt_history_candidate PRIVATE c_std_17)
target_compile_definitions(dango_material_receipt_history_candidate PRIVATE ${repair_definitions} DANGO_RECEIPT_HISTORY_CANDIDATE DANGO_MATERIAL_EXPERIMENT_MODE=15 DANGO_MATERIAL_CONTACT_WIDE_GROUND_CANDIDATE DANGO_MATERIAL_CONTACT_ZERO_TO_SLIDE_CANDIDATE)
target_include_directories(dango_material_receipt_history_candidate BEFORE PRIVATE "${DANGO_RECEIPT_GENERATED}/adapter" "${CMAKE_CURRENT_SOURCE_DIR}" "${DANGO_NATIVE}" "${DANGO_RECEIPT_GENERATED}/box2d/src")
target_link_libraries(dango_material_receipt_history_candidate PRIVATE box2d::box2d)
set_target_properties(dango_material_receipt_history_candidate PROPERTIES PREFIX "")
if(MSVC)
    target_compile_options(dango_material_receipt_history_candidate PRIVATE /W4 /WX)
    target_link_options(dango_material_receipt_history_candidate PRIVATE /Brepro)
endif()
add_executable(dango_receipt_history_tests tests.c capture.c "${DANGO_NATIVE}/dango_material_forces.c")
target_compile_features(dango_receipt_history_tests PRIVATE c_std_17)
target_compile_definitions(dango_receipt_history_tests PRIVATE ${repair_definitions} DANGO_RECEIPT_HISTORY_CANDIDATE DANGO_MATERIAL_EXPERIMENT_MODE=15 DANGO_MATERIAL_CONTACT_WIDE_GROUND_CANDIDATE DANGO_MATERIAL_CONTACT_ZERO_TO_SLIDE_CANDIDATE)
target_include_directories(dango_receipt_history_tests BEFORE PRIVATE "${DANGO_RECEIPT_GENERATED}/adapter" "${CMAKE_CURRENT_SOURCE_DIR}" "${DANGO_NATIVE}" "${DANGO_RECEIPT_GENERATED}/box2d/src")
target_link_libraries(dango_receipt_history_tests PRIVATE box2d::box2d)
if(MSVC)
    target_compile_options(dango_receipt_history_tests PRIVATE /W4 /WX)
endif()
