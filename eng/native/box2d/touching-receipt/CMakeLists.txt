cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(dango_touching_receipt LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BOX2D_VALIDATE ON CACHE BOOL "" FORCE)
add_subdirectory("${DANGO_GENERATED}/box2d" box2d)
target_compile_definitions(box2d PRIVATE DANGO_RECEIPT_HISTORY_CANDIDATE)
target_include_directories(box2d PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../receipt-history")
get_filename_component(DANGO_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
file(STRINGS "${DANGO_NATIVE}/CMakeLists.txt" repair_line REGEX "^set\\(dango_material_repair_final_definitions ")
string(REGEX REPLACE "^set\\(dango_material_repair_final_definitions (.*)\\)$" "\\1" repair_definitions "${repair_line}")
separate_arguments(repair_definitions)
function(configure target)
 target_compile_features(${target} PRIVATE c_std_17)
 target_compile_definitions(${target} PRIVATE ${repair_definitions} DANGO_MATERIAL_EXPERIMENT_MODE=15 DANGO_MATERIAL_CONTACT_WIDE_GROUND_CANDIDATE DANGO_MATERIAL_CONTACT_ZERO_TO_SLIDE_CANDIDATE DANGO_RECEIPT_HISTORY_CANDIDATE)
 target_include_directories(${target} BEFORE PRIVATE "${DANGO_GENERATED}/adapter" "${CMAKE_CURRENT_SOURCE_DIR}" "${DANGO_NATIVE}/receipt-history" "${DANGO_NATIVE}" "${DANGO_GENERATED}/box2d/src")
 target_link_libraries(${target} PRIVATE box2d)
 target_compile_options(${target} PRIVATE /W4 /WX)
endfunction()
add_library(dango_material_touching_receipt_candidate SHARED candidate.c "${DANGO_NATIVE}/receipt-history/capture.c" "${DANGO_NATIVE}/dango_material_forces.c")
configure(dango_material_touching_receipt_candidate)
target_link_options(dango_material_touching_receipt_candidate PRIVATE /Brepro)
foreach(kind IN ITEMS touching receipt endpoint singleton)
 if(kind STREQUAL "touching")
  set(test_source tests.c)
 else()
  set(test_source "${DANGO_NATIVE}/${kind}-history/tests.c")
  if(kind STREQUAL "endpoint")
   set(test_source "${DANGO_NATIVE}/endpoint-selection/tests.c")
  elseif(kind STREQUAL "singleton")
   set(test_source "${DANGO_NATIVE}/singleton-mode/tests.c")
  endif()
 endif()
 add_executable(dango_touching_${kind}_tests "${test_source}" "${DANGO_NATIVE}/receipt-history/capture.c" "${DANGO_NATIVE}/dango_material_forces.c")
 configure(dango_touching_${kind}_tests)
endforeach()
target_link_options(dango_touching_receipt_tests PRIVATE /STACK:8388608)
