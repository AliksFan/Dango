cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(dango_prescribed_position_projection LANGUAGES C)
if(NOT DEFINED BOX2D_SOURCE_DIR OR NOT DEFINED DANGO_NATIVE_SOURCE_DIR)
    message(FATAL_ERROR "Pass verified dependency and native source directories")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX2D_DOCS OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_VALIDATE ON CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${BOX2D_SOURCE_DIR}" box2d EXCLUDE_FROM_ALL)
add_executable(dango_position_projection position_projection.c)
target_link_libraries(dango_position_projection PRIVATE box2d::box2d)
target_include_directories(dango_position_projection PRIVATE "${DANGO_NATIVE_SOURCE_DIR}")
target_compile_features(dango_position_projection PRIVATE c_std_17)
if(MSVC)
    target_compile_options(dango_position_projection PRIVATE /W4 /WX /fp:precise)
    target_link_options(dango_position_projection PRIVATE /Brepro)
endif()
