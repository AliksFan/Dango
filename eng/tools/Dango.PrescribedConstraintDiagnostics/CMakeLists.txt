cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(dango_prescribed_diagnostics LANGUAGES C)
if(NOT DEFINED BOX2D_SOURCE_DIR OR NOT DEFINED DANGO_NATIVE_SOURCE_DIR)
    message(FATAL_ERROR "Pass verified dependency and project native source directories")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX2D_DOCS OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_VALIDATE ON CACHE BOOL "" FORCE)
set(BOX2D_COMPILE_WARNING_AS_ERROR OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${BOX2D_SOURCE_DIR}" box2d EXCLUDE_FROM_ALL)
file(TO_CMAKE_PATH "${DANGO_NATIVE_SOURCE_DIR}" native_path)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiler.json"
    "{\"Id\":\"${CMAKE_C_COMPILER_ID}\",\"Version\":\"${CMAKE_C_COMPILER_VERSION}\",\"Generator\":\"${CMAKE_GENERATOR}\",\"Platform\":\"${CMAKE_GENERATOR_PLATFORM}\",\"ReleaseFlags\":\"${CMAKE_C_FLAGS_RELEASE}\"}\n")
foreach(variant IN ITEMS 0 1 2)
    set(target "dango_constraint_diagnostics_${variant}")
    add_executable(${target} constraint_diagnostics.c)
    target_link_libraries(${target} PRIVATE box2d::box2d)
    target_compile_features(${target} PRIVATE c_std_17)
    target_compile_definitions(${target} PRIVATE DANGO_DIAGNOSTIC_VARIANT=${variant}
        DANGO_DIAGNOSTIC_BRIDGE_SOURCE="${native_path}/dango_box2d_bridge.c"
        DANGO_DIAGNOSTIC_COLLISION_SOURCE="${native_path}/dango_collision_review.c")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /fp:precise)
        target_link_options(${target} PRIVATE /Brepro)
    endif()
endforeach()
