cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(dango_safe_spawn_probe LANGUAGES C)
if(NOT DEFINED BOX2D_SOURCE_DIR)
  message(FATAL_ERROR "BOX2D_SOURCE_DIR must name the existing pinned source")
endif()
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX2D_DOCS OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_VALIDATE ON CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${BOX2D_SOURCE_DIR}" box2d EXCLUDE_FROM_ALL)
add_executable(dango_safe_spawn_probe native_probe.c)
target_link_libraries(dango_safe_spawn_probe PRIVATE box2d::box2d)
target_compile_features(dango_safe_spawn_probe PRIVATE c_std_17)
target_compile_definitions(dango_safe_spawn_probe PRIVATE DANGO_AUTHORITY_CONTRACT_TESTS=1 DANGO_AUTHORITY_TEST_VARIANT=0)
