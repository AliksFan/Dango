cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(dango_physics_replay_admission LANGUAGES C)

if(NOT DEFINED BOX2D_SOURCE_DIR OR NOT DEFINED DANGO_NATIVE_SOURCE_DIR)
    message(FATAL_ERROR "Pass the verified Box2D source and a frozen Dango native source directory")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX2D_DOCS OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_VALIDATE ON CACHE BOOL "" FORCE)
set(BOX2D_COMPILE_WARNING_AS_ERROR OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Replay admission freezes Release" FORCE)
add_subdirectory("${BOX2D_SOURCE_DIR}" box2d EXCLUDE_FROM_ALL)

file(GLOB native_fingerprint_files "${DANGO_NATIVE_SOURCE_DIR}/*.c" "${DANGO_NATIVE_SOURCE_DIR}/*.h")
list(SORT native_fingerprint_files)
set(build_settings "Box2D:8c661469c9507d3ad6fbd2fea3f1aa71669c2fe3;dt:3C888865;${CMAKE_C_COMPILER_ID};${CMAKE_C_COMPILER_VERSION};${CMAKE_GENERATOR};${CMAKE_GENERATOR_PLATFORM};${CMAKE_VS_PLATFORM_TOOLSET};Release;${CMAKE_C_FLAGS};${CMAKE_C_FLAGS_RELEASE};static-runtime;BOX2D_VALIDATE=ON;DANGO_STATIC_PLATFORM_TEST_VARIANT=0;DANGO_ROUND_BALL_TEST_VARIANT=0;W4;WX;fp:precise;Brepro;")
set(fingerprint "${build_settings}")
file(GLOB_RECURSE box2d_fingerprint_files LIST_DIRECTORIES FALSE "${BOX2D_SOURCE_DIR}/*")
list(SORT box2d_fingerprint_files)
set(box2d_fingerprint "")
set(box2d_manifest "")
foreach(source IN LISTS box2d_fingerprint_files)
    file(SHA256 "${source}" source_hash)
    file(RELATIVE_PATH source_name "${BOX2D_SOURCE_DIR}" "${source}")
    string(APPEND box2d_fingerprint "${source_name}:${source_hash};")
    if(box2d_manifest)
        string(APPEND box2d_manifest ",")
    endif()
    string(APPEND box2d_manifest "{\"Path\":\"${source_name}\",\"Sha256\":\"${source_hash}\"}")
endforeach()
string(SHA256 box2d_source_id "${box2d_fingerprint}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/box2d-source-manifest.json" "[${box2d_manifest}]\n")
string(APPEND fingerprint "Box2D actual source:${box2d_source_id};")
set(native_manifest "")
foreach(source IN LISTS native_fingerprint_files)
    file(SHA256 "${source}" source_hash)
    file(RELATIVE_PATH source_name "${DANGO_NATIVE_SOURCE_DIR}" "${source}")
    string(APPEND fingerprint "${source_name}:${source_hash};")
    if(native_manifest)
        string(APPEND native_manifest ",")
    endif()
    string(APPEND native_manifest "{\"Path\":\"${source_name}\",\"Sha256\":\"${source_hash}\"}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/replay_probe.c" probe_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" cmake_hash)
set(build_fingerprint "${fingerprint}probe:${probe_hash};cmake:${cmake_hash};")
string(SHA256 build_id "${build_fingerprint}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build-fingerprint.txt" "${build_fingerprint}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build-identity.json"
    "{\"BuildId\":\"${build_id}\",\"CompilerId\":\"${CMAKE_C_COMPILER_ID}\",\"CompilerVersion\":\"${CMAKE_C_COMPILER_VERSION}\",\"Configuration\":\"Release\",\"ProbeSourceSha256\":\"${probe_hash}\",\"ProbeCMakeSha256\":\"${cmake_hash}\",\"Box2DActualSourceId\":\"${box2d_source_id}\",\"IdentityMethod\":\"SHA256 of build-fingerprint.txt containing ordered actual source names and hashes, probe and CMake hashes, dependency source digest and frozen build settings\",\"NativeSources\":[${native_manifest}]}\n")
file(TO_CMAKE_PATH "${DANGO_NATIVE_SOURCE_DIR}" native_path)
add_executable(dango_physics_replay_probe replay_probe.c)
target_link_libraries(dango_physics_replay_probe PRIVATE box2d::box2d)
target_compile_features(dango_physics_replay_probe PRIVATE c_std_17)
target_compile_definitions(dango_physics_replay_probe PRIVATE
    DANGO_REPLAY_BUILD_ID="${build_id}"
    DANGO_REPLAY_BRIDGE_SOURCE="${native_path}/dango_box2d_bridge.c"
    DANGO_REPLAY_COLLISION_SOURCE="${native_path}/dango_collision_review.c"
    DANGO_STATIC_PLATFORM_TEST_VARIANT=0 DANGO_ROUND_BALL_TEST_VARIANT=0)
if(MSVC)
    target_compile_options(dango_physics_replay_probe PRIVATE /W4 /WX /fp:precise)
    target_link_options(dango_physics_replay_probe PRIVATE /Brepro)
endif()
